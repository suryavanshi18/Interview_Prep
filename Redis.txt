##########################################

from fastapi import FastAPI
import uvicorn
from redis import Redis
from rq import Queue
from pydantic import BaseModel
from job import print_number 
class JobPayload(BaseModel):
    lowest:int
    highest:int
    

app=FastAPI()
redis_con=Redis(host='localhost',port=6379)
task_queue=Queue("task_queue",connection=redis_con)
@app.get("/")
def index():
    return {"success":True,"message":"pong"}

@app.post("/job")
def post_job(payload:JobPayload):
    lowest=payload.lowest
    highest=payload.highest
    
    job_instance=task_queue.enqueue(print_number,lowest,highest)
    return {"success":True,
            "job":job_instance.id}

##########################################
def print_number(lowest,highest):
    x=lowest
    print("== Job processing")
    while(x<=highest):
        print("{}\n".format(x))
        x=x+1
        
    print("== End")
